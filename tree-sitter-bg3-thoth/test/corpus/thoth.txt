local __util = require 'larian.util'

function Calculate(entity, fallback)
  entity = entity or context.Source
  local result = fallback or 10
  if entity.IsValid then
    try
      result = CalculateSpellDC(entity)
    catch error then
      result = fallback
    end
  end
  return result
end

function CanApply(entity)
  return HasStatus('TEST', entity) & ~Dead(entity)
end

function SilentFallback(entity)
  try
    return CalculateSpellDC(entity)
  catch error then
  end
  return 10
end
